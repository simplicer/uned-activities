### ITERACIÓN AUTOMÁTICA 14: Improve Harvest Loop with Retry Logic
**Objective:** Add exponential backoff and error handling
**File:** `/infra/scripts/harvest-loop.sh` (rewrite)
**Content:**
```bash
#!/bin/bash

set -euo pipefail

LOG_FILE="/data/logs/harvest.log"
mkdir -p "$(dirname "$LOG_FILE")"

# Configuration
HARVEST_URL="${HARVEST_URL:-http://api:8080/v1/jobs/harvest}"
MAX_RETRIES=5
INITIAL_DELAY=5
MAX_DELAY=3600

# Logging function
log_message() {
    local level=$1
    local message=$2
    echo "[$(date -u +'%Y-%m-%dT%H:%M:%SZ')] [$level] $message" | tee -a "$LOG_FILE"
}

# Retry with exponential backoff
retry_harvest() {
    local attempt=1
    local delay=$INITIAL_DELAY
    
    while [ $attempt -le $MAX_RETRIES ]; do
        log_message "INFO" "Harvest attempt $attempt/$MAX_RETRIES"
        
        if curl -f -X POST "$HARVEST_URL" \
            -H "Content-Type: application/json" \
            -d '{"force":true}' \
            --connect-timeout 10 \
            --max-time 600 \
            >> "$LOG_FILE" 2>&1; then
            
            log_message "INFO" "Harvest completed successfully"
            return 0
        fi
        
        if [ $attempt -lt $MAX_RETRIES ]; then
            log_message "WARN" "Harvest failed, retrying in ${delay}s (attempt $attempt/$MAX_RETRIES)"
            sleep "$delay"
            
            # Exponential backoff: min 5s, max 1h
            delay=$((delay * 2))
            if [ $delay -gt $MAX_DELAY ]; then
                delay=$MAX_DELAY
            fi
        fi
        
        attempt=$((attempt + 1))
    done
    
    log_message "ERROR" "Harvest failed after $MAX_RETRIES attempts"
    return 1
}

# Main loop
log_message "INFO" "Harvest loop started"

while true; do
    retry_harvest || log_message "ERROR" "Giving up on this cycle"
    
    # Sleep 1 hour before next harvest attempt
    log_message "INFO" "Sleeping 1 hour until next harvest cycle"
    sleep 3600
done
```

### ITERACIÓN AUTOMÁTICA 17: Create Migration Runner Script
**Objective:** PHP script to manage migrations
**File:** `/infra/scripts/migrate.php` (rewrite for idempotency)
```php
<?php
// Migration runner - idempotent
// Usage: php migrate.php [up|down]

$direction = $argv[1] ?? 'up';
$migrationsDir = __DIR__ . '/../migrations';

$pdo = new PDO(
    "pgsql:host={$_ENV['DB_HOST']};dbname={$_ENV['DB_NAME']}",
    $_ENV['DB_USER'],
    $_ENV['DB_PASSWORD']
);

// Create migrations table if not exists
$pdo->exec("
    CREATE TABLE IF NOT EXISTS migrations (
        id SERIAL PRIMARY KEY,
        name VARCHAR(255) UNIQUE NOT NULL,
        executed_at TIMESTAMP DEFAULT NOW()
    )
");

if ($direction === 'up') {
    // Get all migration files
    $files = glob("$migrationsDir/*.up.sql");
    sort($files);
    
    foreach ($files as $file) {
        $name = basename($file);
        
        // Check if already executed
        $stmt = $pdo->prepare("SELECT id FROM migrations WHERE name = ?");
        if ($stmt->execute([$name])->fetch()) {
            echo "[SKIP] $name (already executed)\n";
            continue;
        }
        
        // Execute migration
        $sql = file_get_contents($file);
        $pdo->exec($sql);
        
        // Record execution
        $pdo->prepare("INSERT INTO migrations (name) VALUES (?)")->execute([$name]);
        echo "[OK] $name\n";
    }
} elseif ($direction === 'down') {
    // Get last executed migration
    $stmt = $pdo->query("SELECT name FROM migrations ORDER BY id DESC LIMIT 1");
    if ($row = $stmt->fetch()) {
        $migration = $row['name'];
        $rollbackFile = str_replace('.up.sql', '.down.sql', "$migrationsDir/$migration");
        
        if (file_exists($rollbackFile)) {
            $sql = file_get_contents($rollbackFile);
            $pdo->exec($sql);
            $pdo->prepare("DELETE FROM migrations WHERE name = ?")->execute([$migration]);
            echo "[OK] Rolled back: $migration\n";
        } else {
            echo "[ERROR] No rollback script for $migration\n";
            exit(1);
        }
    }
}
```

**Commit Message:** `feat(infra): create idempotent migration runner script`
**Verification:** `grep -q "CREATE TABLE IF NOT EXISTS migrations" infra/scripts/migrate.php && echo "✓"`
